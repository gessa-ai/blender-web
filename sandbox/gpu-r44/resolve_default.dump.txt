vec3 workbench_normal_decode(vec4 enc)
{
  vec2 fenc = enc.xy * 4.0f - 2.0f;
  float f = dot(fenc, fenc);
  float g = sqrt(1.0f - f / 4.0f);
  vec3 n;
  n.xy = fenc * g;
  n.z = 1 - f / 2;
  return n;
}























void workbench_float_pair_decode(float data, inout float v1, inout float v2)
{

  const int v1_mask = 0x1F;
  const int v2_mask = 0x7;
  int idata = int(data);
  v1 = float(idata & v1_mask) * (1.0f / float(v1_mask));
  v2 = float(idata >> int((8u - 3u))) * (1.0f / float(v2_mask));
}

vec4 workbench_fast_rcp(vec4 v)
{
  return intBitsToFloat(0x7eef370b - floatBitsToInt(v));
}

vec3 workbench_brdf_approx(vec3 spec_color, float roughness, float NV)
{

  float fresnel = exp2(-8.35f * NV) * (1.0f - roughness);
  return mix(spec_color, vec3(1.0f), fresnel);
}

vec4 workbench_blinn_specular(vec4 shininess, vec4 spec_angle, vec4 NL)
{

  vec4 normalization_factor = shininess * 0.125f + 1.0f;
  vec4 spec_light = pow(spec_angle, shininess) * NL * normalization_factor;

  return spec_light;
}

vec4 workbench_wrapped_lighting(vec4 NL, vec4 w)
{
  vec4 w_1 = w + 1.0f;
  vec4 denom = workbench_fast_rcp(w_1 * w_1);
  return clamp((NL + w) * denom, 0.0f, 1.0f);
}



vec3 workbench_get_world_lighting(const workbench_World  world,
                          vec3 base_color,
                          float roughness,
                          float metallic,
                          vec3 N,
                          vec3 I)
{

  vec3 specular_color, diffuse_color;

  if (world_data.use_specular) {
    diffuse_color = mix(base_color, vec3(0.0f), metallic);
    specular_color = mix(vec3(0.05f), base_color, metallic);
  }
  else {
    diffuse_color = base_color;
    specular_color = vec3(0.0f);
  }

  vec3 specular_light = world_data.ambient_color.rgb;
  vec3 diffuse_light = world_data.ambient_color.rgb;
  vec4 wrap = vec4(world_data.lights[0].diffuse_color_wrap.a,
                       world_data.lights[1].diffuse_color_wrap.a,
                       world_data.lights[2].diffuse_color_wrap.a,
                       world_data.lights[3].diffuse_color_wrap.a);

  if (world_data.use_specular) {

    vec3 R = -reflect(I, N);

    vec4 spec_angle, spec_NL, wrapped_NL;

{

                                           {
      vec3 L = world_data.lights[0].direction.xyz;
      vec3 half_dir = normalize(L + I);
      wrapped_NL[0] = dot(L, R);
      spec_angle[0] = clamp(dot(half_dir, N), 0.0f, 1.0f);
      spec_NL[0] = clamp(dot(L, N), 0.0f, 1.0f);
    }

                                           {
      vec3 L = world_data.lights[1].direction.xyz;
      vec3 half_dir = normalize(L + I);
      wrapped_NL[1] = dot(L, R);
      spec_angle[1] = clamp(dot(half_dir, N), 0.0f, 1.0f);
      spec_NL[1] = clamp(dot(L, N), 0.0f, 1.0f);
    }

                                           {
      vec3 L = world_data.lights[2].direction.xyz;
      vec3 half_dir = normalize(L + I);
      wrapped_NL[2] = dot(L, R);
      spec_angle[2] = clamp(dot(half_dir, N), 0.0f, 1.0f);
      spec_NL[2] = clamp(dot(L, N), 0.0f, 1.0f);
    }

                                           {
      vec3 L = world_data.lights[3].direction.xyz;
      vec3 half_dir = normalize(L + I);
      wrapped_NL[3] = dot(L, R);
      spec_angle[3] = clamp(dot(half_dir, N), 0.0f, 1.0f);
      spec_NL[3] = clamp(dot(L, N), 0.0f, 1.0f);
    }

    }

    vec4 gloss = vec4(1.0f - roughness);

    gloss *= 1.0f - wrap;
    vec4 shininess = exp2(10.0f * gloss + 1.0f);

    vec4 spec_light = workbench_blinn_specular(shininess, spec_angle, spec_NL);

    vec4 w = mix(wrap, vec4(1.0f), roughness);
    vec4 spec_env = workbench_wrapped_lighting(wrapped_NL, w);

    spec_light = mix(spec_light, spec_env, wrap * wrap);

{

                                           {
      specular_light += spec_light[0] * world_data.lights[0].specular_color.rgb;
    }

                                           {
      specular_light += spec_light[1] * world_data.lights[1].specular_color.rgb;
    }

                                           {
      specular_light += spec_light[2] * world_data.lights[2].specular_color.rgb;
    }

                                           {
      specular_light += spec_light[3] * world_data.lights[3].specular_color.rgb;
    }

    }

    float NV = clamp(dot(N, I), 0.0f, 1.0f);
    specular_color = workbench_brdf_approx(specular_color, roughness, NV);
  }
  specular_light *= specular_color;

  vec4 diff_NL;

{

                                         {
    diff_NL[0] = dot(world_data.lights[0].direction.xyz, N);
  }

                                         {
    diff_NL[1] = dot(world_data.lights[1].direction.xyz, N);
  }

                                         {
    diff_NL[2] = dot(world_data.lights[2].direction.xyz, N);
  }

                                         {
    diff_NL[3] = dot(world_data.lights[3].direction.xyz, N);
  }

  }

  vec4 diff_light = workbench_wrapped_lighting(diff_NL, wrap);

{

                                         {
    diffuse_light += diff_light[0] * world_data.lights[0].diffuse_color_wrap.rgb;
  }

                                         {
    diffuse_light += diff_light[1] * world_data.lights[1].diffuse_color_wrap.rgb;
  }

                                         {
    diffuse_light += diff_light[2] * world_data.lights[2].diffuse_color_wrap.rgb;
  }

                                         {
    diffuse_light += diff_light[3] * world_data.lights[3].diffuse_color_wrap.rgb;
  }

  }

  float spec_energy = dot(specular_color, vec3(0.33333f));

  diffuse_light *= diffuse_color * (1.0f - spec_energy);

  return diffuse_light + specular_light;
}
